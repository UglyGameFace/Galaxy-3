cmake_minimum_required(VERSION 3.16)
project(uglyos-app-hub VERSION 0.1 DESCRIPTION "UgLy|OS offline native app catalog")
find_package(ECM REQUIRED NO_MODULE)
find_package(AsteroidApp6 REQUIRED)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH} ${ASTEROID_MODULE_PATH})
include(FeatureSummary)
include(GNUInstallDirs)
include(Asteroid6CMakeSettings)
include(Asteroid6Translations)
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/uglyos-app-hub.in
               ${CMAKE_BINARY_DIR}/uglyos-app-hub @ONLY)
install(PROGRAMS ${CMAKE_BINARY_DIR}/uglyos-app-hub
        DESTINATION ${CMAKE_INSTALL_BINDIR})
add_library(uglyos-app-hub main.cpp)
set_target_properties(uglyos-app-hub PROPERTIES PREFIX "")
asteroid_app_qml_module(uglyos-app-hub main.qml HubPage.qml HubCatalog.js)
target_link_libraries(uglyos-app-hub PUBLIC Asteroid::AsteroidApp6)
install(TARGETS uglyos-app-hub DESTINATION ${CMAKE_INSTALL_LIBDIR})
generate_desktop(${CMAKE_SOURCE_DIR} uglyos-app-hub)
feature_summary(WHAT ALL FATAL_ON_MISSING_REQUIRED_PACKAGES)
