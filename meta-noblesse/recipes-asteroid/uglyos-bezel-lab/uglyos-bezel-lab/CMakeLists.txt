cmake_minimum_required(VERSION 3.16)
project(uglyos-bezel-lab VERSION 0.1 DESCRIPTION "UgLy|OS offline native bezel interaction lab")
find_package(ECM REQUIRED NO_MODULE)
find_package(AsteroidApp6 REQUIRED)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH} ${ASTEROID_MODULE_PATH})
include(FeatureSummary)
include(GNUInstallDirs)
include(Asteroid6CMakeSettings)
include(Asteroid6Translations)
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/uglyos-bezel-lab.in
               ${CMAKE_BINARY_DIR}/uglyos-bezel-lab @ONLY)
install(PROGRAMS ${CMAKE_BINARY_DIR}/uglyos-bezel-lab
        DESTINATION ${CMAKE_INSTALL_BINDIR})
add_library(uglyos-bezel-lab main.cpp)
set_target_properties(uglyos-bezel-lab PROPERTIES PREFIX "")
asteroid_app_qml_module(uglyos-bezel-lab main.qml BezelPage.qml BezelEngine.js)
target_link_libraries(uglyos-bezel-lab PUBLIC Asteroid::AsteroidApp6)
install(TARGETS uglyos-bezel-lab DESTINATION ${CMAKE_INSTALL_LIBDIR})
generate_desktop(${CMAKE_SOURCE_DIR} uglyos-bezel-lab)
feature_summary(WHAT ALL FATAL_ON_MISSING_REQUIRED_PACKAGES)
