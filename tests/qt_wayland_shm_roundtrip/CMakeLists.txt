cmake_minimum_required(VERSION 3.16)
project(watch3_wayland_shm_roundtrip LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Widgets WaylandCompositor)
add_executable(watch3_shm_compositor compositor.cpp)
target_link_libraries(watch3_shm_compositor PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::WaylandCompositor)
add_executable(watch3_shm_client client.cpp)
target_link_libraries(watch3_shm_client PRIVATE Qt6::Widgets)
